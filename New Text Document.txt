    header("Location: index.php");

$admin_id = $_SESSION['user_id']; // Session එකේ ඇති Admin ID එක
$msg = "";

// දැනට ඇති විස්තර ලබා ගැනීම
$query = mysqli_query($conn, "SELECT username, email FROM users WHERE id = '$admin_id'");
$admin_data = mysqli_fetch_assoc($query);

// Update Logic
if (isset($_POST['update_profile'])) {
    $new_email = mysqli_real_escape_string($conn, $_POST['email']);
    $new_pass = $_POST['password'];

    if (!empty($new_pass)) {
        // Password එකක් ඇතුළත් කර ඇත්නම් එය Hash කර Update කිරීම
        $hashed_pass = password_hash($new_pass, PASSWORD_DEFAULT);
        $update_query = "UPDATE users SET email='$new_email', password='$hashed_pass' WHERE id='$admin_id'";
    } else {
        // Password එක වෙනස් නොකරන්නේ නම් Email එක පමණක් Update කිරීම
        $update_query = "UPDATE users SET email='$new_email' WHERE id='$admin_id'";
    }

    if (mysqli_query($conn, $update_query)) {
        $msg = "<div class='alert-success'>Profile updated successfully!</div>";
        // අලුත් Email එක ලබා ගැනීමට Refresh කිරීම
        $admin_data['email'] = $new_email;
    } else {
        $msg = "<div class='alert-danger'>Error updating profile.</div>";
    }
}
